cmake_minimum_required(VERSION 3.16)
project(siegebreaker VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(CURL 7.85 REQUIRED)
find_package(Threads REQUIRED)
set(SANITIZER "" CACHE STRING "address, undefined or thread")
if(SANITIZER)
  if(NOT SANITIZER MATCHES "^(address|undefined|thread)$")
    message(FATAL_ERROR "Unsupported sanitizer")
  endif()
  add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=${SANITIZER} -no-pie)
endif()
add_library(hdr STATIC vendor/hdr/hdr_histogram.c)
target_include_directories(hdr PUBLIC vendor/hdr/include)
target_link_libraries(hdr PUBLIC m)
add_library(core src/core.c src/worker.c)
target_include_directories(core PUBLIC src)
target_compile_definitions(core PUBLIC _POSIX_C_SOURCE=200809L)
target_compile_options(core PUBLIC -Wall -Wextra -Wpedantic)
target_link_libraries(core PUBLIC CURL::libcurl hdr Threads::Threads)
add_executable(siegebreaker src/main.c)
target_link_libraries(siegebreaker PRIVATE core)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(unit tests/unit.c)
  target_link_libraries(unit PRIVATE core)
  add_test(NAME unit COMMAND unit)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:siegebreaker>)
  set_tests_properties(integration PROPERTIES TIMEOUT 45)
endif()
